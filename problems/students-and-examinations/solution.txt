SELECT st.student_id, st.student_name, sb.subject_name, COUNT(e.student_id) as attended_exams
CROSS JOIN Subjects AS sb
LEFT JOIN Examinations AS e
ON st.student_id = e.student_id AND
e.subject_name = sb.subject_name
FROM Students AS st
GROUP BY st.student_name, st.student_id, sb.subject_name
ORDER BY st.student_id, sb.subject_name;
